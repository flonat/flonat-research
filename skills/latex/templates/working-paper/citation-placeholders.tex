% citation-placeholders.tex — BACK-COMPAT SHIM (2026-07-24).
%
% The \CiteTodo guard now ships inside flonat-bib.sty, so new papers need only:
%   \usepackage[harvard]{flonat-bib}      % guard included, fail-closed
%   \usepackage[nocitetodo]{flonat-bib}   % opt out
%
% This file remains so existing papers that \input it keep compiling. It defines
% the guard only when flonat-bib has not already done so, and is a no-op
% otherwise (loading both in either order is safe).
%
% Draft mode (placeholders render as visible red flags instead of erroring):
%   \draftcitationstodotrue   % AFTER this file / flonat-bib is loaded

\expandafter\ifx\csname ifdraftcitationstodo\endcsname\relax
  \expandafter\newif\csname ifdraftcitationstodo\endcsname
\fi

\providecommand{\CiteTodo}[2]{%
  \ifdraftcitationstodo
    \textbf{[UNRESOLVED CITATION: #1 --- #2]}%
  \else
    \PackageError{paperpile-citations}{Unresolved citation `#1': #2}{%
      Resolve via Paperpile/refpile (paperpile export-bib). Do not invent a citekey.%
    }%
  \fi
}
